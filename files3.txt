@RestController("pmHandoverPendingController")
@RequestMapping("/api/pm/handover")
@RequiredArgsConstructor
public class HandoverPendingController {

    public record PendingRow(Long employeeId, String name, LocalDate lastWorkingDay,
                             int openTasks, int ownedProjects, int ledTeams) {}
    public record Candidate(Long id, String name) {}

    private final PmHandoverFinder finder;
    private final EmployeeRepository employeeRepo;
    private final PmAccessService access;

    private boolean privileged(Long userId) { return access.isAdmin(userId) || access.isHr(userId); }   // ADAPT isHr

    /** Admin/HR see everything; a manager sees only the parts in projects they administer. */
    private PmHandoverFinder.Detail visible(PmHandoverFinder.Detail d, Long userId) {
        if (d == null || privileged(userId)) return d;
        var tasks = d.tasks().stream().filter(t -> access.canAdministerProject(userId, t.projectId())).collect(Collectors.toList());
        var projects = d.ownedProjects().stream().filter(p -> access.canAdministerProject(userId, p.id())).collect(Collectors.toList());
        var teams = access.canManageTeamLeads(userId) ? d.ledTeams() : List.<PmHandoverFinder.Ref>of();        // ADAPT
        return new PmHandoverFinder.Detail(d.employeeId(), d.name(), d.lastWorkingDay(), tasks, projects, teams);
    }

    private boolean empty(PmHandoverFinder.Detail d) {
        return d.tasks().isEmpty() && d.ownedProjects().isEmpty() && d.ledTeams().isEmpty();
    }

    @GetMapping("/pending")
    public ResponseEntity<ApiResponse<List<PendingRow>>> pending(@CurrentUserId Long userId) {
        List<PendingRow> out = new ArrayList<>();
        for (var e : employeeRepo.findByStatusIn(List.of("RESIGNED", "NOTICE_PERIOD"))) {       // ADAPT repo method + values
            var d = visible(finder.detail(e.getId()), userId);
            if (d == null || empty(d)) continue;
            out.add(new PendingRow(d.employeeId(), d.name(), d.lastWorkingDay(),
                    d.tasks().size(), d.ownedProjects().size(), d.ledTeams().size()));
        }
        out.sort(Comparator.comparing(PendingRow::lastWorkingDay, Comparator.nullsLast(Comparator.naturalOrder())));
        return ResponseEntity.ok(ApiResponse.ok(out));
    }

    @GetMapping("/pending/{employeeId}")
    public ResponseEntity<ApiResponse<PmHandoverFinder.Detail>> detail(
            @PathVariable Long employeeId, @CurrentUserId Long userId) {
        var d = visible(finder.detail(employeeId), userId);
        if (d == null) throw new ResourceNotFoundException("Employee not found: " + employeeId);
        if (!privileged(userId) && empty(d)) throw new PmAccessDeniedException("You have nothing to hand over for this employee.");
        return ResponseEntity.ok(ApiResponse.ok(d));
    }

    /** Active employees for the pickers on the handover and team-lead pages. */
    @GetMapping("/candidates")
    public ResponseEntity<ApiResponse<List<Candidate>>> candidates(@CurrentUserId Long userId) {
        if (!privileged(userId) && !access.canManageTeamLeads(userId) && !access.isAnyProjectManager(userId))   // ADAPT both
            throw new PmAccessDeniedException("Not allowed.");
        List<Candidate> out = employeeRepo.findAll().stream()
                .filter(e -> "ACTIVE".equals(e.getStatus()))                                                    // ADAPT
                .map(e -> new Candidate(e.getId(),
                        ((e.getFirstName() == null ? "" : e.getFirstName()) + " " + (e.getLastName() == null ? "" : e.getLastName())).trim()))
                .sorted(Comparator.comparing(Candidate::name, String.CASE_INSENSITIVE_ORDER))
                .collect(Collectors.toList());
        return ResponseEntity.ok(ApiResponse.ok(out));
    }

@RestController("pmTaskBulkController")

public class TaskBulkController {

    private final TaskService taskService;
    private final PmAccessService access;

    @PostMapping("/filter")
    public ResponseEntity<ApiResponse<PageResponse<TaskSummaryResponse>>> filter(
            @RequestBody TaskFilterRequest f, @CurrentUserId Long userId) {
        if (f.getProjectId() == null) throw new BusinessRuleException("projectId is required");
        access.requireCanViewProject(userId, f.getProjectId());
        return ResponseEntity.ok(ApiResponse.ok(taskService.filterTasks(f)));
    }

    /**
     * Not one big transaction: each task goes through the normal changeStatus / assignTask
     * (workflow rules, bells, audit log) on its own, so one bad task never blocks the rest.
     */
    @PostMapping("/bulk")
    public ResponseEntity<ApiResponse<Map<String, Object>>> bulk(
            @Valid @RequestBody BulkTaskRequest req, @CurrentUserId Long userId) {

        String action = req.getAction().trim().toUpperCase();
        Set<String> ACTIONS = Set.of("STATUS", "ASSIGN", "ADD_ASSIGNEE", "PRIORITY", "SPRINT", "DELETE");
        if (!ACTIONS.contains(action)) throw new BusinessRuleException("Unknown action: " + action);
        if (action.equals("STATUS") && req.getStatus() == null)
            throw new BusinessRuleException("status is required for the STATUS action");
        if ((action.equals("ASSIGN") || action.equals("ADD_ASSIGNEE"))
                && (req.getEmployeeIds() == null || req.getEmployeeIds().isEmpty()))
            throw new BusinessRuleException("employeeIds is required for assigning");
        if (action.equals("PRIORITY") && req.getPriority() == null)
            throw new BusinessRuleException("priority is required");                       // ADAPT: add to BulkTaskRequest
// SPRINT: sprintId may be null = move to backlog

        int ok = 0;
        List<String> failed = new ArrayList<>();
        List<String> warnings = new ArrayList<>();

        for (Long id : new LinkedHashSet<>(req.getTaskIds())) {
            try {
                TaskResponse cur = taskService.getTask(id);
                if (access.isProjectLocked(cur.getProjectId()))
                    throw new BusinessRuleException("Project is completed or archived, so its tasks are locked");
                switch (action) {
                    case "STATUS" -> {
                        if (!access.canEditTaskAccess(userId, id)) throw new PmAccessDeniedException("You can't change this task");
                        TaskStatusChangeRequest r = new TaskStatusChangeRequest();
                        r.setStatus(req.getStatus()); r.setReason(req.getReason());
                        taskService.changeStatus(id, r, userId);
                    }
                    case "PRIORITY", "SPRINT" -> {
                        if (!access.canEditTaskAccess(userId, id)) throw new PmAccessDeniedException("You can't change this task");
                        UpdateTaskRequest u = new UpdateTaskRequest();                     // ADAPT setter names
                        if (action.equals("PRIORITY")) u.setPriority(req.getPriority());
                        else u.setSprintId(req.getSprintId() == null ? 0L : req.getSprintId());   // ADAPT: 0 = clear, if your update uses that rule
                        taskService.updateTask(id, u, userId);
                    }
                    case "DELETE" -> {
                        if (!access.canEditTaskAccess(userId, id)) throw new PmAccessDeniedException("You can't delete this task");
                        taskService.deleteTask(id, userId);
                    }
                    default -> {
                        if (!access.canAssignTask(userId, id)) throw new PmAccessDeniedException("You can't assign this task");
                        LinkedHashSet<Long> ids = new LinkedHashSet<>();
                        if (action.equals("ADD_ASSIGNEE") && cur.getAssigneeIds() != null) ids.addAll(cur.getAssigneeIds());
                        ids.addAll(req.getEmployeeIds());
                        AssignTaskRequest r = new AssignTaskRequest();
                        r.setEmployeeIds(new ArrayList<>(ids));
                        TaskResponse res = taskService.assignTask(id, r, userId);
                        if (res.getAssigneeIds() != null && res.getAssignmentWarnings() != null)
                            warnings.addAll(res.getAssignmentWarnings());
                    }
                }
                ok++;
            } catch (Exception e) {
                failed.add("Task #" + id + ": " + e.getMessage());
            }
        }

        Map<String, Object> out = new LinkedHashMap<>();
        out.put("ok", ok);
        out.put("failed", failed);
        out.put("warnings", new LinkedHashSet<>(warnings));
        return ResponseEntity.ok(ApiResponse.ok("Bulk action finished", out));
    }
}

package com.mentis.hrms.pm.dto.request;

import com.fasterxml.jackson.annotation.JsonIgnoreProperties;
import com.mentis.hrms.pm.enums.ProjectPriority;
import com.mentis.hrms.pm.enums.TaskPriority;
import com.mentis.hrms.pm.enums.TaskType;
import lombok.Getter;
import lombok.NoArgsConstructor;
import lombok.Setter;

import java.util.ArrayList;
import java.util.List;

/**
 * Typed shape of pm_project_templates.payload:
 * { "project": {"name","priority"}, "milestones":[{"name","dueOffsetDays"}],
 *   "sprints":[{"name","goal","startOffsetDays","durationDays"}],
 *   "tasks":[{"title","priority","type","assignToRole","estimateHours","storyPoints",
 *             "dueOffsetDays","sprintIndex","milestoneIndex"}] }
 * Offsets are days from the project start date. Indexes are 0-based positions in the lists above.
 */
@Getter @Setter @NoArgsConstructor
@JsonIgnoreProperties(ignoreUnknown = true)
public class ProjectTemplatePayload {

    public static final int MAX_MILESTONES = 30, MAX_SPRINTS = 10, MAX_TASKS = 100;

    private ProjectPart project = new ProjectPart();
    private List<MilestonePart> milestones = new ArrayList<>();
    private List<SprintPart> sprints = new ArrayList<>();
    private List<TaskPart> tasks = new ArrayList<>();
    public static final int MAX_PHASES = 15;
    private List<PhasePart> phases = new ArrayList<>();
    @Getter @Setter @NoArgsConstructor @JsonIgnoreProperties(ignoreUnknown = true)
    public static class ProjectPart { private String name; private String priority; }

    @Getter @Setter @NoArgsConstructor @JsonIgnoreProperties(ignoreUnknown = true)
    public static class MilestonePart { private String name; private Integer dueOffsetDays; }
    @Getter @Setter @NoArgsConstructor @JsonIgnoreProperties(ignoreUnknown = true)
    public static class PhasePart { private String name; private Integer startOffsetDays; private Integer durationDays; }
    @Getter @Setter @NoArgsConstructor @JsonIgnoreProperties(ignoreUnknown = true)
    public static class SprintPart {
        private String name; private String goal; private Integer startOffsetDays; private Integer durationDays;
    }

    @Getter @Setter @NoArgsConstructor @JsonIgnoreProperties(ignoreUnknown = true)
    public static class TaskPart {
        private String title; private String priority; private String type; private String assignToRole;
        private Integer estimateHours; private Integer storyPoints; private Integer dueOffsetDays;
        private Integer sprintIndex; private Integer milestoneIndex;
    }

    /** Throws IllegalArgumentException (400) with a message that says exactly what is wrong. */
    public void validate() {
        if (project == null) project = new ProjectPart();
        if (milestones == null) milestones = new ArrayList<>();
        if (sprints == null) sprints = new ArrayList<>();
        if (tasks == null) tasks = new ArrayList<>();
        if (phases == null) phases = new ArrayList<>();
        if (phases.size() > MAX_PHASES) bad("A template can have at most " + MAX_PHASES + " phases");
        for (int i = 0; i < phases.size(); i++) {
            PhasePart ph = phases.get(i);
            String f = "phases[" + i + "]";
            text(ph.getName(), f + ".name", 200);
            range(ph.getStartOffsetDays(), 0, 3650, f + ".startOffsetDays", true);
            range(ph.getDurationDays(), 1, 730, f + ".durationDays", true);
        }
        if (milestones.size() > MAX_MILESTONES) bad("A template can have at most " + MAX_MILESTONES + " milestones");
        if (sprints.size() > MAX_SPRINTS) bad("A template can have at most " + MAX_SPRINTS + " sprints");
        if (tasks.size() > MAX_TASKS) bad("A template can have at most " + MAX_TASKS + " tasks");

        enumOk(project.getPriority(), ProjectPriority.class, "project.priority");

        for (int i = 0; i < milestones.size(); i++) {
            MilestonePart m = milestones.get(i);
            String f = "milestones[" + i + "]";
            text(m.getName(), f + ".name", 200);
            range(m.getDueOffsetDays(), 0, 3650, f + ".dueOffsetDays", true);
        }

        int prevEndExclusive = 0; // next sprint may start on this offset or later (touching dates overlap)
        for (int i = 0; i < sprints.size(); i++) {
            SprintPart s = sprints.get(i);
            String f = "sprints[" + i + "]";
            text(s.getName(), f + ".name", 200);
            range(s.getStartOffsetDays(), 0, 3650, f + ".startOffsetDays", true);
            range(s.getDurationDays(), 1, 90, f + ".durationDays", true);
            if (s.getStartOffsetDays() < prevEndExclusive)
                bad(f + " overlaps the previous sprint (start at day " + prevEndExclusive + " or later)");
            prevEndExclusive = s.getStartOffsetDays() + s.getDurationDays();
        }

        for (int i = 0; i < tasks.size(); i++) {
            TaskPart t = tasks.get(i);
            String f = "tasks[" + i + "]";
            text(t.getTitle(), f + ".title", 300);
            enumOk(t.getPriority(), TaskPriority.class, f + ".priority");
            enumOk(t.getType(), TaskType.class, f + ".type");
            if (t.getType() != null && "SUBTASK".equalsIgnoreCase(t.getType().trim()))
                bad(f + ".type cannot be SUBTASK (subtasks need a parent task)");
            if (t.getAssignToRole() != null && !t.getAssignToRole().isBlank()) {
                String r = t.getAssignToRole().trim().toUpperCase();
                if (!r.equals("MANAGER") && !r.equals("MEMBER"))
                    bad(f + ".assignToRole must be MANAGER or MEMBER");
            }
            range(t.getEstimateHours(), 0, 1000, f + ".estimateHours", false);
            range(t.getStoryPoints(), 0, 100, f + ".storyPoints", false);
            range(t.getDueOffsetDays(), 0, 3650, f + ".dueOffsetDays", false);
            if (t.getSprintIndex() != null && (t.getSprintIndex() < 0 || t.getSprintIndex() >= sprints.size()))
                bad(f + ".sprintIndex points to a sprint that does not exist");
            if (t.getMilestoneIndex() != null && (t.getMilestoneIndex() < 0 || t.getMilestoneIndex() >= milestones.size()))
                bad(f + ".milestoneIndex points to a milestone that does not exist");
        }
    }

    private static void bad(String m) { throw new IllegalArgumentException(m); }

    private static void text(String v, String field, int max) {
        if (v == null || v.isBlank()) bad(field + " is required");
        if (v.trim().length() > max) bad(field + " is too long (max " + max + " characters)");
    }

    private static void range(Integer v, int min, int max, String field, boolean required) {
        if (v == null) { if (required) bad(field + " is required"); return; }
        if (v < min || v > max) bad(field + " must be between " + min + " and " + max);
    }

    private static <E extends Enum<E>> void enumOk(String v, Class<E> type, String field) {
        if (v == null || v.isBlank()) return;
        try { Enum.valueOf(type, v.trim().toUpperCase()); }
        catch (IllegalArgumentException e) { bad(field + " has an unknown value: " + v); }
    }
}

    public void applyCompletionTransition(TaskStatus from, TaskStatus to) {

            this.completedAt = LocalDateTime.now();
        } else if (from == TaskStatus.DONE && to != TaskStatus.DONE) {
            this.completedAt = null;
        }
    }

        if (priority == null) priority = TaskPriority.MEDIUM;
        if (type == null) type = TaskType.TASK;
        if (loggedHours == null) loggedHours = 0;
        if (loggedMinutes == null) loggedMinutes = 0;

    private final TaskRepository taskRepo;
    private final TaskTimeLogRepository logRepo;
    private final ProjectBaselineHistoryRepository histRepo;
    private final PmAccessService access;

    @Transactional(readOnly = true)
    public PlanActualResponse get(Long projectId, Long userId) {
        access.requireCanViewProject(userId, projectId);
        if (!access.canViewFinance(userId, projectId))                 // ADAPT: your finance-read check
            throw new PmAccessDeniedException("You do not have access to Plan vs Actual for this project.");

        Project p = projectRepo.findById(projectId)
                .orElseThrow(() -> new ResourceNotFoundException("Project not found: " + projectId));
        LocalDate today = LocalDate.now(ZoneId.of("Asia/Kolkata"));
        boolean baselined = p.getBaselinedAt() != null;

        List<Task> tasks = taskRepo.findByProjectId(projectId);        // ADAPT: your repo method
        Map<Long, Double> logged = new HashMap<>();
        if (!tasks.isEmpty()) {
            for (Object[] o : logRepo.sumMinutesByTask(tasks.stream().map(Task::getId).collect(Collectors.toList())))
                logged.put((Long) o[0], ((Number) o[1]).doubleValue() / 60.0);
        }

        List<PlanActualResponse.Row> rows = new ArrayList<>();
        int doneWithBaseline = 0, onTime = 0, late = 0, added = 0, slipN = 0;
        double slipSum = 0, baseHoursSum = 0, loggedHoursSum = 0;

        for (Task t : tasks) {
            LocalDate bd = t.getBaselineDueDate();                     // ADAPT field
            Double bh = t.getBaselineHours();                          // ADAPT field (Double or BigDecimal -> convert)
            LocalDate done = (t.getStatus() == TaskStatus.DONE && t.getCompletedAt() != null)
                    ? t.getCompletedAt().toLocalDate() : null;         // ADAPT: LocalDateTime or LocalDate
            Integer slip = bd == null ? null : (int) ChronoUnit.DAYS.between(bd, done != null ? done : today);
            double lh = logged.getOrDefault(t.getId(), 0.0);
            boolean addedAfter = baselined && t.getCreatedAt() != null
                    && t.getCreatedAt().isAfter(p.getBaselinedAt());   // ADAPT types

            if (bd != null && done != null) { doneWithBaseline++; if (!done.isAfter(bd)) onTime++; }
            if (slip != null && (done != null || today.isAfter(bd))) { slipSum += Math.max(0, slip); slipN++; }
            if (slip != null && slip > 0) late++;
            if (addedAfter) added++;
            if (bh != null && bh > 0) { baseHoursSum += bh; loggedHoursSum += lh; }

            rows.add(PlanActualResponse.Row.builder()
                    .taskId(t.getId()).title(t.getTitle()).status(t.getStatus().name())
                    .baselineDue(bd).currentDue(t.getDueDate()).completedOn(done)
                    .slipDays(slip).baselineHours(bh).loggedHours(Math.round(lh * 10) / 10.0)
                    .addedAfterBaseline(addedAfter).build());
        }

        PlanActualResponse.Summary s = PlanActualResponse.Summary.builder()
                .onTimePct(doneWithBaseline == 0 ? null : (int) Math.round(onTime * 100.0 / doneWithBaseline))
                .avgSlipDays(slipN == 0 ? null : Math.round(slipSum / slipN * 10) / 10.0)
                .hoursVariancePct(baseHoursSum <= 0 ? null
                        : Math.round((loggedHoursSum - baseHoursSum) / baseHoursSum * 1000) / 10.0)
                .lateTasks(late).addedAfterBaseline(added).build();

        boolean locked = access.isProjectLocked(projectId);
        return PlanActualResponse.builder()
                .projectId(projectId).baselined(baselined).baselinedAt(p.getBaselinedAt())
                .baselineVersion((int) histRepo.countByProjectId(projectId) + (baselined ? 1 : 0))   // ADAPT
                .locked(locked)
                .canManage(!locked && access.canAdministerProject(userId, projectId))
                .summary(s).rows(rows).build();
    }

                .stream().map(this::toSummaryWithNames).collect(Collectors.toList());

        res.setAssignmentWarnings(warnings);
        return res;
    }

    @Override

    public void deleteTask(Long id, Long currentUserId) {
        Task task = taskRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException("Task not found: " + id));
        Long parentId = task.getParentTaskId();
        Long projectId = task.getProjectId();

        long subs = taskRepository.countSubtasks(id);
        if (subs > 0)
            throw new BusinessRuleException("Delete or move the " + subs + " subtask(s) first");

        taskAttachmentRepository.deleteByTaskId(id);
        taskTimeLogRepository.deleteByTaskId(id);
        taskAssignmentRepository.deleteByTaskId(id);
        taskWatcherRepository.deleteByTaskId(id);
        taskCommentRepository.deleteAll(taskCommentRepository.findByTaskIdAndDeletedFalseOrderByCreatedAtAsc(id));
        checklistItemRepository.deleteByTaskId(id);
        taskDependencyRepository.deleteByTaskId(id);
        taskDependencyRepository.deleteByDependsOnTaskId(id);
        taskRepository.deleteById(id);

        if (parentId != null) {
            taskRepository.findById(parentId).ifPresent(this::rollUp);
        } else if (projectId != null) {
            projectRepository.findById(projectId).ifPresent(p -> {
                p.setProgressPct(avg(taskRepository.findByProjectIdAndParentTaskIdIsNull(p.getId())));
                projectRepository.save(p);
            });
        }
        log.info("Task deleted id={}", id);
    }

        if (!mentions.isBlank()) {

                    String recipientCode = resolveEmployeeCode(mentionPk);
                    if (recipientCode == null) continue;

                    notificationEmitter.emit(NotificationRequest.builder()
                            .recipientId(recipientCode)
                            .recipientType("EMPLOYEE")
                            .type("TASK_MENTIONED")
                            .action("TASK_MENTIONED")

                            .projectId(projectId)
                            .actorId(userId)
                            .actorName(actorName2)
                            .priority("HIGH")

                            .build());
                } catch (Exception e) {
                    log.warn("Failed to emit TASK_MENTIONED for {}: {}", idStr, e.getMessage());
                }
            }
        }
        return toCommentResponse(saved);
    }

    // ─── Step 3.2B — include project owner + team members in mention pool ───
    private String parseMentions(Long taskId, String body) {
        if (body == null || body.isBlank()) return "";
        java.util.regex.Matcher m = java.util.regex.Pattern
                .compile("@([A-Za-z0-9._]+)").matcher(body);
        java.util.Set<String> handles = new java.util.LinkedHashSet<>();
        while (m.find()) handles.add(m.group(1).toLowerCase());
        if (handles.isEmpty()) return "";

        Long projectId = taskRepository.findById(taskId)
                .map(Task::getProjectId).orElse(null);
        if (projectId == null) return "";

        java.util.Set<Long> memberIds = new java.util.HashSet<>();
        taskRepository.findByProjectId(projectId).forEach(t ->
                taskAssignmentRepository.findByTaskId(t.getId())
                        .forEach(a -> memberIds.add(a.getEmployeeId())));
        projectMemberRepository.findByProjectId(projectId)
                .forEach(pm -> memberIds.add(pm.getEmployeeId()));
        projectRepository.findById(projectId).ifPresent(p -> {
            if (p.getOwnerId() != null) memberIds.add(p.getOwnerId());
            if (p.getTeamId() != null)
                teamMemberRepository.findByTeamId(p.getTeamId())
                        .forEach(tm -> memberIds.add(tm.getEmployeeId()));
        });

        java.util.List<String> matched = new java.util.ArrayList<>();
        for (Long empId : memberIds) {
            String name = resolveEmployeeName(empId);
            if (name == null) continue;
            String normalized = name.toLowerCase().replaceAll("\\s+", "");
            String firstName = name.split("\\s+")[0].toLowerCase();
            if (handles.contains(normalized) || handles.contains(firstName)) {
                matched.add(String.valueOf(empId));
            }
        }
        return String.join(",", matched);
    }

    @Override
    @Transactional(readOnly = true)
    public List<ChecklistItemResponse> listChecklist(Long taskId) {
        List<TaskChecklistItem> all = checklistItemRepository.findByTaskIdOrderBySortOrderAsc(taskId);
        log.info("listChecklist taskId={} found {} rows", taskId, all.size());

        Map<Long, List<TaskChecklistItem>> byParent = new HashMap<>();
        for (TaskChecklistItem item : all) {
            Long key = item.getParentId();
            byParent.computeIfAbsent(key, k -> new ArrayList<>()).add(item);
        }

        List<ChecklistItemResponse> roots = byParent.getOrDefault(null, List.of()).stream()
                .map(item -> buildTree(item, byParent))
                .collect(Collectors.toList());

        for (TaskChecklistItem item : all) {
            if (item.getParentId() != null && !all.stream().anyMatch(x -> x.getId().equals(item.getParentId()))) {
                roots.add(buildTree(item, byParent));
            }
        }

        return roots;
    }

    private ChecklistItemResponse buildTree(TaskChecklistItem item,
                                            Map<Long, List<TaskChecklistItem>> byParent) {
        List<ChecklistItemResponse> children = byParent.getOrDefault(item.getId(), List.of()).stream()
                .map(c -> buildTree(c, byParent))
                .collect(Collectors.toList());
        return ChecklistItemResponse.builder()
                .id(item.getId()).taskId(item.getTaskId()).parentId(item.getParentId())
                .title(item.getTitle()).done(item.getDone())
                .assigneeId(item.getAssigneeId()).sortOrder(item.getSortOrder())
                .createdAt(item.getCreatedAt()).children(children).build();
    }

    // ─── Step 1.4 — addDependency with cycle guard ───
    @Override
    @Transactional
    public void addDependency(Long taskId, AddDependencyRequest req) {
        Long dependsOnId = req.getDependsOnTaskId();
        if (!taskRepository.existsById(taskId))
            throw new ResourceNotFoundException("Task not found: " + taskId);
        if (dependsOnId == null || !taskRepository.existsById(dependsOnId))
            throw new ResourceNotFoundException("Depends-on task not found: " + dependsOnId);
        if (taskId.equals(dependsOnId))
            throw new BusinessRuleException("A task cannot depend on itself");

        boolean duplicate = taskDependencyRepository.findByTaskId(taskId).stream()
                .anyMatch(d -> dependsOnId.equals(d.getDependsOnTaskId()));
        if (duplicate)
            throw new BusinessRuleException("This dependency already exists");

        if (wouldCreateCycle(taskId, dependsOnId))
            throw new BusinessRuleException(
                    "This would create a circular dependency: the other task already "
                            + "depends on this one, directly or through other tasks");

        TaskDependency dep = taskDependencyRepository.save(TaskDependency.builder()
                .taskId(taskId)
                .dependsOnTaskId(dependsOnId)
                .type(req.getType() != null ? req.getType() : "BLOCKED_BY")
                .build());
        log.info("Dependency added id={} task={} dependsOn={} type={}",
                dep.getId(), taskId, dependsOnId, dep.getType());
        logActivity(taskId, "DEPENDENCY_ADDED", null, null,
                String.valueOf(dependsOnId), "Dependency added");
    }

    /**
     * Edge direction in your tables: row(taskId = X, dependsOnTaskId = Y) means
     * "X is blocked by Y". Adding X -> Y is a cycle if Y can already reach X.
     */
    private boolean wouldCreateCycle(Long taskId, Long dependsOnId) {
        Deque<Long> stack = new ArrayDeque<>();
        stack.push(dependsOnId);
        Set<Long> seen = new HashSet<>();
        while (!stack.isEmpty()) {
            Long cur = stack.pop();
            if (cur.equals(taskId)) return true;
            if (!seen.add(cur)) continue;
            for (TaskDependency d : taskDependencyRepository.findByTaskId(cur)) {
                stack.push(d.getDependsOnTaskId());
            }
        }
        return false;
    }

    private String describeChanges(Map<String, String> before, Map<String, String> after) {
        List<String> parts = new ArrayList<>();
        after.forEach((k, v) -> {
            String b = before.get(k);
            if (!Objects.equals(b, v)) parts.add(k + ": " + b + " → " + v);
        });
        return parts.isEmpty() ? "Task updated (no field changes)" : String.join("; ", parts);
    }

    private TaskResponse toFullResponse(Task task) {
        List<Long> ids = taskAssignmentRepository.findByTaskId(task.getId())
                .stream().map(TaskAssignment::getEmployeeId).collect(Collectors.toList());
        List<String> names = resolveEmployeeNames(ids);
        String teamName = task.getTeamId() != null
                ? teamRepository.findById(task.getTeamId()).map(Team::getName).orElse(null)
                : null;
        String projectName = task.getProjectId() != null
                ? projectRepository.findById(task.getProjectId()).map(Project::getName).orElse(null)
                : null;

        String sprintName = null;
        if (task.getSprintId() != null) {
            try {
                sprintName = sprintRepository.findById(task.getSprintId())
                        .map(Sprint::getName).orElse(null);
            } catch (Exception ex) {
                log.warn("Sprint {} missing for task {}: {}", task.getSprintId(), task.getId(), ex.getMessage());
            }
        }

        String milestoneName = null;
        if (task.getMilestoneId() != null) {
            try {
                milestoneName = milestoneRepository.findById(task.getMilestoneId())
                        .map(Milestone::getName).orElse(null);
            } catch (Exception ex) {
                log.warn("Milestone {} missing for task {}: {}", task.getMilestoneId(), task.getId(), ex.getMessage());
            }
        }

        TaskResponse res = taskMapper.toResponse(
                task,
                task.getTeamId(), teamName,
                task.getProjectId(), projectName,
                ids, names,
                task.getSprintId(), sprintName,
                task.getMilestoneId(), milestoneName
        );

        res.setSubtaskCount(taskRepository.countSubtasks(task.getId()));
        res.setSubtaskDone(taskRepository.countSubtasksByStatus(task.getId(), TaskStatus.DONE));
        res.setOpenBlockerCount(taskRepository.countOpenBlockers(task.getId(), TaskStatus.DONE));

        return res;
    }

                : null;

        if (t.getSprintId() != null) {
            try {
                sprintName = sprintRepository.findById(t.getSprintId())
                        .map(Sprint::getName).orElse(null);
            } catch (Exception ex) { /* ignore */ }

    private List<String> resolveEmployeeNames(List<Long> ids) {
        if (ids == null || ids.isEmpty()) return List.of();
        List<String> out = new ArrayList<>(ids.size());
        for (Long id : ids) {
            try {
                out.add(resolveEmployeeName(id));
            } catch (Exception ex) {
                out.add("Employee #" + id);
            }
        }
        return out;
    }

    // ─── Step 2.5 — block direct DONE when a sign-off is pending ───
    private void assertCanMoveTo(Task task, TaskStatus target) {
        if (target != TaskStatus.DONE) return;

        if (approvalRequestRepository.existsByEntityTypeAndEntityIdAndStatus("TASK", task.getId(), "PENDING"))
            throw new BusinessRuleException(
                    "This task is waiting for sign-off. It becomes DONE when the approver approves.");

        long blockers = taskRepository.countOpenBlockers(task.getId(), TaskStatus.DONE);
        if (blockers > 0)
            throw new BusinessRuleException("Can't mark DONE: " + blockers + " open blocker(s)");
        long subs = taskRepository.countSubtasks(task.getId());
        long done = taskRepository.countSubtasksByStatus(task.getId(), TaskStatus.DONE);
        if (subs > done)
            throw new BusinessRuleException("Can't mark DONE: " + (subs - done) + " subtask(s) still open");
        long openChecklist = checklistItemRepository.findByTaskIdOrderBySortOrderAsc(task.getId()).stream()
                .filter(i -> !Boolean.TRUE.equals(i.getDone())).count();
        if (openChecklist > 0)
            throw new BusinessRuleException("Can't mark DONE: " + openChecklist + " checklist item(s) still open");
    }
    /** Allowed-moves rule + "don't walk away from a pending sign-off". */
    /** Allowed-moves rule (incl. Step 4 review gate) + "don't walk away from a pending sign-off". */
    private void assertTransition(Task task, TaskStatus to, Long userId) {
        TaskStatus from = task.getStatus();
        if (from == to) return;
        boolean gate = pmAccessService.isReviewGateActive(userId, task.getId());
        if (!TaskWorkflow.allowed(from, to, gate))
            throw new BusinessRuleException("A task cannot move from " + TaskWorkflow.label(from)
                    + " to " + TaskWorkflow.label(to) + "." + TaskWorkflow.hint(from, to, gate));
        if (from == TaskStatus.IN_REVIEW && to != TaskStatus.DONE
                && approvalRequestRepository.existsByEntityTypeAndEntityIdAndStatus("TASK", task.getId(), "PENDING"))
            throw new BusinessRuleException(
                    "This task is waiting for sign-off. Cancel the approval request first.");
    }


    private void rollUp(Task t) {
        if (t.getParentTaskId() != null) {
            taskRepository.findById(t.getParentTaskId()).ifPresent(parent -> {
                parent.setProgressPct(avg(taskRepository.findSubtasks(parent.getId())));
                taskRepository.save(parent);
                rollUp(parent);
            });
            return;
        }
        projectRepository.findById(t.getProjectId()).ifPresent(p -> {
            p.setProgressPct(avg(taskRepository.findByProjectIdAndParentTaskIdIsNull(p.getId())));
            projectRepository.save(p);
        });
    }

    // ─── Step 2.6 — weighted progress ───
    private double weightOf(Task t) {
        Integer est = t.getEstimateHours();
        if (est != null && est > 0) return est;
        Integer sp = t.getStoryPoints();
        if (sp != null && sp > 0) return sp * 4.0;
        return 1.0;
    }

    private int avg(List<Task> l) {
        double w = 0, s = 0;
        for (Task t : l) {
            double x = weightOf(t);
            int p = t.getProgressPct() == null ? 0 : t.getProgressPct();
            w += x;
            s += x * p;
        }
        return w == 0 ? 0 : (int) Math.round(s / w);
    }

    private String resolveEmployeeName(Long id) {
        if (id == null) return null;
        try {
            return employeeRepository.findById(id)
                    .map(e -> {
                        try {
                            String first = e.getFirstName() == null ? "" : e.getFirstName();
                            String last  = e.getLastName()  == null ? "" : e.getLastName();
                            String full = (first + " " + last).trim();
                            if (!full.isEmpty()) return full;
                            if (e.getEmployeeName() != null && !e.getEmployeeName().isBlank())
                                return e.getEmployeeName();
                            if (e.getEmail() != null) return e.getEmail();
                        } catch (Exception ignored) {
                        }
                        return "Employee #" + id;
                    })
                    .orElse("Employee #" + id);
        } catch (Exception ex) {
            log.warn("resolveEmployeeName failed for id={}: {}", id, ex.getMessage());
            return "Employee #" + id;
        }
    }

    private String resolveEmployeeCode(Long employeePk) {
        if (employeePk == null) return null;
        try {
            return employeeRepository.findById(employeePk)
                    .map(e -> e.getEmployeeId())
                    .orElse(null);
        } catch (Exception ex) {
            log.warn("resolveEmployeeCode failed for pk={}: {}", employeePk, ex.getMessage());
            return null;
        }
    }

    private String trimPreview(String s, int max) {
        if (s == null) return "";
        String cleaned = s.replaceAll("\\s+", " ").trim();
        return cleaned.length() <= max ? cleaned : cleaned.substring(0, max) + "…";
    }
