package com.mentis.hrms.config;

import jakarta.servlet.http.HttpSession;
import lombok.extern.slf4j.Slf4j;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.server.ServerHttpRequest;
import org.springframework.http.server.ServerHttpResponse;
import org.springframework.http.server.ServletServerHttpRequest;
import org.springframework.messaging.Message;
import org.springframework.messaging.MessageChannel;
import org.springframework.messaging.simp.config.ChannelRegistration;
import org.springframework.messaging.simp.config.MessageBrokerRegistry;
import org.springframework.messaging.simp.stomp.StompCommand;
import org.springframework.messaging.simp.stomp.StompHeaderAccessor;
import org.springframework.messaging.support.ChannelInterceptor;
import org.springframework.messaging.support.MessageHeaderAccessor;
import org.springframework.web.socket.WebSocketHandler;
import org.springframework.web.socket.config.annotation.*;
import org.springframework.web.socket.server.HandshakeInterceptor;

import java.security.Principal;
import java.util.Map;

@Configuration
@EnableWebSocketMessageBroker
@Slf4j
public class WebSocketConfig implements WebSocketMessageBrokerConfigurer {

    @Override
    public void configureMessageBroker(MessageBrokerRegistry config) {
        config.enableSimpleBroker("/topic", "/queue", "/user");
        config.setApplicationDestinationPrefixes("/app");
        config.setUserDestinationPrefix("/user");
    }

    @Override
    public void registerStompEndpoints(StompEndpointRegistry registry) {
        registry.addEndpoint("/ws")
                .setAllowedOriginPatterns("*")
                .addInterceptors(new SessionPrincipalHandshakeInterceptor())
                .withSockJS()
                .setStreamBytesLimit(512 * 1024)
                .setHttpMessageCacheSize(1000)
                .setDisconnectDelay(30 * 1000);

        registry.addEndpoint("/ws-native")
                .setAllowedOriginPatterns("*")
                .addInterceptors(new SessionPrincipalHandshakeInterceptor());
    }

    /**
     * Reads HttpSession attribute "userId" (set by the HRMS login)
     * and attaches it as a Principal on the WebSocket session.
     */
    public static class SessionPrincipalHandshakeInterceptor implements HandshakeInterceptor {
        @Override
        public boolean beforeHandshake(ServerHttpRequest request, ServerHttpResponse response,
                                       WebSocketHandler wsHandler, Map<String, Object> attributes) {
            if (request instanceof ServletServerHttpRequest sreq) {
                HttpSession session = sreq.getServletRequest().getSession(false);
                if (session != null) {
                    Object uid = session.getAttribute("userId");
                    if (uid != null) {
                        attributes.put("userId", String.valueOf(uid));
                    }
                }
            }
            // permit anonymous handshake so SockJS info requests still work;
            // the inbound interceptor below rejects subscriptions without a Principal
            return true;
        }

        @Override
        public void afterHandshake(ServerHttpRequest request, ServerHttpResponse response,
                                   WebSocketHandler wsHandler, Exception exception) { }
    }

    /**
     * Plain Principal â€” no Spring Security dependency.
     * STOMP only needs getName(); we don't need authorities here because
     * message-level auth is handled by PmAccessService at the REST layer.
     */
    private static final class SimplePrincipal implements Principal {
        private final String name;
        SimplePrincipal(String name) { this.name = name; }
        @Override public String getName() { return name; }
        @Override public String toString() { return name; }
    }

    /**
     * Attaches a Principal on CONNECT from the handshake attributes and
     * blocks subscriptions to another user's personal topic.
     */
    @Override
    public void configureClientInboundChannel(ChannelRegistration registration) {
        registration.interceptors(new ChannelInterceptor() {
            @Override
            public Message<?> preSend(Message<?> message, MessageChannel channel) {
                StompHeaderAccessor acc =
                        MessageHeaderAccessor.getAccessor(message, StompHeaderAccessor.class);
                if (acc == null) return message;

                // Attach principal on CONNECT from handshake attributes
                if (StompCommand.CONNECT.equals(acc.getCommand())
                        && acc.getSessionAttributes() != null) {
                    Object uid = acc.getSessionAttributes().get("userId");
                    if (uid != null) {
                        acc.setUser(new SimplePrincipal(String.valueOf(uid)));
                    }
                }

                // Authorize /topic/user/{id}/... subscriptions
                if (StompCommand.SUBSCRIBE.equals(acc.getCommand())) {
                    String dest = acc.getDestination();
                    if (dest != null && dest.startsWith("/topic/user/")) {
                        String[] parts = dest.split("/");
                        // /topic/user/{userId}/notifications â†’ parts[3]
                        if (parts.length >= 4) {
                            String requested = parts[3];
                            Principal me = acc.getUser();
                            if (me == null || !me.getName().equals(requested)) {
                                log.warn("Blocked topic subscription: {} (me={})", dest,
                                        me == null ? "anon" : me.getName());
                                return null; // silently drop the frame
                            }
                        }
                    }
                }
                return message;
            }
        });
    }
}

@RestController("pmSavedFilterController")
@RequestMapping("/api/pm/saved-filters")
@RequiredArgsConstructor
public class SavedFilterController {

    public record Body(String name, Long projectId, JsonNode criteria) {}
    public record RenameBody(String name) {}
    public record Item(Long id, String name, Long projectId, JsonNode criteria, boolean isDefault) {}

    private final SavedFilterRepository repo;
    private final PmAccessService access;
    private final ObjectMapper mapper;

    @GetMapping
    public ResponseEntity<ApiResponse<List<Item>>> mine(
            @RequestParam(required = false) Long projectId, @CurrentUserId Long userId) {
        List<Item> out = repo.findByOwnerIdOrderByNameAsc(userId).stream()
                .filter(f -> f.getProjectId() == null || f.getProjectId().equals(projectId))
                .map(this::toItem).collect(Collectors.toList());
        return ResponseEntity.ok(ApiResponse.ok(out));
    }

    @PostMapping
    public ResponseEntity<ApiResponse<Item>> save(@RequestBody Body b, @CurrentUserId Long userId) {
        String name = b.name() == null ? "" : b.name().trim();
        if (name.isEmpty() || name.length() > 100) throw new BusinessRuleException("Name is required (max 100 characters)");
        if (b.criteria() == null || !b.criteria().isObject()) throw new BusinessRuleException("criteria must be a JSON object");
        String json = b.criteria().toString();
        if (json.length() > 5000) throw new BusinessRuleException("This filter is too large");
        if (b.projectId() != null) access.requireCanViewProject(userId, b.projectId());

        SavedFilter f = repo.findByOwnerIdAndProjectIdAndName(userId, b.projectId(), name)
                .orElseGet(() -> SavedFilter.builder().ownerId(userId).projectId(b.projectId()).name(name).build());
        f.setCriteria(json);
        return ResponseEntity.ok(ApiResponse.ok("Saved", toItem(repo.save(f))));
    }

    @PutMapping("/{id}")
    public ResponseEntity<ApiResponse<Item>> rename(@PathVariable Long id, @RequestBody RenameBody b,
                                                    @CurrentUserId Long userId) {
        SavedFilter f = mine(id, userId);
        String name = b.name() == null ? "" : b.name().trim();
        if (name.isEmpty() || name.length() > 100) throw new BusinessRuleException("Name is required (max 100 characters)");
        repo.findByOwnerIdAndProjectIdAndName(userId, f.getProjectId(), name)
                .filter(x -> !x.getId().equals(id))
                .ifPresent(x -> { throw new BusinessRuleException("You already have a filter with that name"); });
        f.setName(name);
        return ResponseEntity.ok(ApiResponse.ok("Renamed", toItem(repo.save(f))));
    }

    @PostMapping("/{id}/default")
    @Transactional
    public ResponseEntity<ApiResponse<Item>> setDefault(@PathVariable Long id, @CurrentUserId Long userId) {
        SavedFilter f = mine(id, userId);
        for (SavedFilter x : repo.findByOwnerIdAndProjectId(userId, f.getProjectId())) {
            if (x.isDefault() && !x.getId().equals(id)) {
                x.setDefault(false);
                repo.save(x);
            }
        }
        f.setDefault(true);
        return ResponseEntity.ok(ApiResponse.ok("Default set", toItem(repo.save(f))));
    }

    @DeleteMapping("/{id}/default")
    public ResponseEntity<ApiResponse<Item>> clearDefault(@PathVariable Long id, @CurrentUserId Long userId) {
        SavedFilter f = mine(id, userId);
        f.setDefault(false);
        return ResponseEntity.ok(ApiResponse.ok("Default cleared", toItem(repo.save(f))));
    }

    @DeleteMapping("/{id}")
    public ResponseEntity<ApiResponse<Void>> delete(@PathVariable Long id, @CurrentUserId Long userId) {
        SavedFilter f = repo.findById(id).orElseThrow(() -> new ResourceNotFoundException("Filter not found: " + id));
        if (!Objects.equals(f.getOwnerId(), userId)) throw new PmAccessDeniedException("You can only delete your own filters.");
        repo.delete(f);
        return ResponseEntity.ok(ApiResponse.ok("Deleted", null));
    }

    private SavedFilter mine(Long id, Long userId) {
        SavedFilter f = repo.findById(id).orElseThrow(() -> new ResourceNotFoundException("Filter not found: " + id));
        if (!Objects.equals(f.getOwnerId(), userId)) throw new PmAccessDeniedException("You can only change your own filters.");
        return f;
    }

    private Item toItem(SavedFilter f) {
        JsonNode node;
        try {
            node = mapper.readTree(f.getCriteria());
        } catch (Exception e) {
            node = mapper.createObjectNode();
        }
        return new Item(f.getId(), f.getName(), f.getProjectId(), node, f.isDefault());
    }
}

package com.mentis.hrms.pm.service;

import com.mentis.hrms.pm.exception.BusinessRuleException;
import java.util.*;

public final class TemplateGraphValidator {

    static final int MAX_TASKS = 500, MAX_CHECKLIST = 50, MAX_ITEM_LEN = 300, MAX_DEPS = 20, MAX_KEY_LEN = 40;

    private TemplateGraphValidator() {}

    private static <T> List<T> nz(List<T> l) { return l == null ? List.of() : l; }

    /** Cleans checklists in place and throws BusinessRuleException on the first problem. */
    public static void validate(List<TaskPart> parts) {              // ADAPT: import TaskPart
        if (parts.size() > MAX_TASKS)
            throw new BusinessRuleException("A template can hold at most " + MAX_TASKS + " tasks.");

        Set<String> keys = new HashSet<>();
        for (TaskPart t : parts) {
            if (t.getKey() != null) {
                String k = t.getKey().trim();
                if (k.isEmpty() || k.length() > MAX_KEY_LEN)
                    throw new BusinessRuleException("A task key must be 1-" + MAX_KEY_LEN + " characters.");
                if (!keys.add(k)) throw new BusinessRuleException("Duplicate task key: " + k);
                t.setKey(k);
            }
            List<String> clean = new ArrayList<>();
            for (String item : nz(t.getChecklist())) {
                String s = item == null ? "" : item.trim();
                if (s.isEmpty()) continue;
                if (s.length() > MAX_ITEM_LEN)
                    throw new BusinessRuleException("A checklist item is longer than " + MAX_ITEM_LEN + " characters.");
                clean.add(s);
            }
            if (clean.size() > MAX_CHECKLIST)
                throw new BusinessRuleException("A task can have at most " + MAX_CHECKLIST + " checklist items.");
            t.setChecklist(clean);
        }

        Map<String, List<String>> graph = new HashMap<>();
        for (TaskPart t : parts) {
            List<String> deps = nz(t.getDependsOn());
            if (deps.isEmpty()) { if (t.getKey() != null) graph.put(t.getKey(), List.of()); continue; }
            if (t.getKey() == null) throw new BusinessRuleException("A task with dependencies needs a key.");
            if (deps.size() > MAX_DEPS) throw new BusinessRuleException("A task can depend on at most " + MAX_DEPS + " others.");
            Set<String> seen = new LinkedHashSet<>();
            for (String d : deps) {
                if (d == null || !keys.contains(d.trim())) throw new BusinessRuleException("Unknown dependency: " + d);
                if (d.trim().equals(t.getKey())) throw new BusinessRuleException("A task cannot depend on itself: " + d);
                seen.add(d.trim());
            }
            t.setDependsOn(new ArrayList<>(seen));
            graph.put(t.getKey(), t.getDependsOn());
        }

        // cycle check: DFS, state 0 = new, 1 = on the current path, 2 = finished
        Map<String, Integer> state = new HashMap<>();
        for (String k : graph.keySet()) visit(k, graph, state);
    }

    private static void visit(String k, Map<String, List<String>> g, Map<String, Integer> state) {
        int s = state.getOrDefault(k, 0);
        if (s == 2) return;
        if (s == 1) throw new BusinessRuleException("Dependency cycle through task " + k);
        state.put(k, 1);
        for (String d : g.getOrDefault(k, List.of())) visit(d, g, state);
        state.put(k, 2);
    }
}

    @Transactional
    public Map<String, Object> applyLeave(LeaveRequest leaveRequest) {
        Map<String, Object> result = new HashMap<>();

        if (leaveRequest.getEmployeeId() == null || leaveRequest.getEmployeeId().isEmpty()) {
            result.put("success", false);
            result.put("error", "Employee ID is required");
            return result;
        }
        if (leaveRequest.getStartDate() == null || leaveRequest.getEndDate() == null) {
            result.put("success", false);
            result.put("error", "Start date and end date are required");
            return result;
        }
        if (leaveRequest.getEndDate().isBefore(leaveRequest.getStartDate())) {
            result.put("success", false);
            result.put("error", "End date cannot be before start date");
            return result;
        }
        if (leaveRequest.getLeaveType() == null || leaveRequest.getLeaveType().trim().isEmpty()) {
            result.put("success", false);
            result.put("error", "Leave type is required (sick, casual, or earned)");
            return result;
        }

        boolean hasSufficientBalance = leaveBalanceService.hasSufficientBalance(
                leaveRequest.getEmployeeId(),
                leaveRequest.getLeaveType(),
                leaveRequest.getStartDate(),
                leaveRequest.getEndDate(),
                leaveRequest.getLeaveDuration(),
                leaveRequest.getHalfDayType()
        );

        if (!hasSufficientBalance) {
            result.put("success", false);
            result.put("error", "Insufficient leave balance");
            return result;
        }

        if (leaveRequest.getTotalDays() == null || leaveRequest.getTotalDays() == 0) {
            long days = ChronoUnit.DAYS.between(leaveRequest.getStartDate(), leaveRequest.getEndDate()) + 1;
            leaveRequest.setTotalDays((double) days);
        }

        leaveRequest.setStatus("PENDING");
        leaveRequest.setCreatedAt(LocalDateTime.now());

        try {
            LeaveRequest saved = leaveRequestRepository.save(leaveRequest);
            logger.info("Leave request {} created for employee {}", saved.getId(), saved.getEmployeeId());

            result.put("success", true);
            result.put("message", "Leave request submitted successfully");
            result.put("leaveId", saved.getId());
            result.put("status", saved.getStatus());
        } catch (Exception e) {
            logger.error("Error saving leave request: {}", e.getMessage(), e);
            result.put("success", false);
            result.put("error", "Failed to save to database: " + e.getMessage());
        }
        return result;
    }

    // ---------------------------------------------------------------
    // APPROVE - PENDING -> APPROVED, deducts balance, creates attendance
    // ---------------------------------------------------------------
    @Transactional
    public Map<String, Object> approveLeave(String leaveId, String approverId, String remarks) {
        Map<String, Object> result = new HashMap<>();
        try {
            LeaveRequest leave = leaveRequestRepository.findById(leaveId)
                    .orElseThrow(() -> new RuntimeException("Leave not found: " + leaveId));

            if ("APPROVED".equals(leave.getStatus())) {
                result.put("success", false);
                result.put("error", "Leave " + leaveId + " is already approved");
                return result;
            }
            if (!"PENDING".equals(leave.getStatus())) {
                result.put("success", false);
                result.put("error", "Only PENDING leaves can be approved (current status: " + leave.getStatus() + ")");
                return result;
            }

            if (leave.getTotalDays() == null || leave.getTotalDays() == 0) {
                long days = ChronoUnit.DAYS.between(leave.getStartDate(), leave.getEndDate()) + 1;
                leave.setTotalDays((double) days);
                leave.setLeaveDuration("full_day");
            }

            leave.setStatus("APPROVED");
            leave.setApprovedBy(approverId != null ? approverId : "SYSTEM");
            leave.setApprovedDate(LocalDateTime.now());
            if (remarks != null && !remarks.trim().isEmpty()) {
                leave.setApprovalRemarks(remarks);
            }

            boolean deducted = leaveBalanceService.deductLeaveBalance(leave);
            if (!deducted) {
                logger.warn("Could not deduct leave balance for {} during approval - check balance records", leaveId);
            }

            LeaveRequest saved = leaveRequestRepository.saveAndFlush(leave);

            createAttendanceForApprovedLeave(saved);

            // GAP 4 FIX: Trigger Notification safely AFTER save
            notifyProjectManagersOfLeave(saved);

            result.put("success", true);
            result.put("message", "Leave approved successfully");
            result.put("leaveId", leaveId);
            result.put("status", saved.getStatus());
        } catch (Exception e) {
            logger.error("Error approving leave {}: {}", leaveId, e.getMessage(), e);
            result.put("success", false);
            result.put("error", e.getMessage());
        }
        return result;
    }

    @Transactional
    public Map<String, Object> cancelLeave(String leaveId, String reason) {
        Map<String, Object> result = new HashMap<>();
        try {
            LeaveRequest leave = leaveRequestRepository.findById(leaveId)
                    .orElseThrow(() -> new RuntimeException("Leave not found: " + leaveId));

            String previousStatus = leave.getStatus();

            if (!"PENDING".equals(previousStatus) && !"APPROVED".equals(previousStatus)) {
                result.put("success", false);
                result.put("error", "Only PENDING or APPROVED leaves can be cancelled (current status: " + previousStatus + ")");
                return result;
            }

            leave.setStatus("CANCELLED");
            leave.setCancelReason(reason);
            leave.setWithdrawn(true);
            leave.setWithdrawnDate(LocalDateTime.now());

            boolean balanceRestored = false;
            if ("APPROVED".equals(previousStatus)) {
                balanceRestored = leaveBalanceService.addBackLeaveBalance(leave);
                if (!balanceRestored) {
                    logger.warn("Could not restore leave balance for cancelled leave {}", leaveId);
                }
            }

            LeaveRequest saved = leaveRequestRepository.save(leave);

            result.put("success", true);
            result.put("message", "Leave cancelled successfully"
                    + (balanceRestored ? " and leave balance restored" : ""));
            result.put("leaveId", leaveId);
            result.put("status", saved.getStatus());
        } catch (Exception e) {
            logger.error("Error cancelling leave {}: {}", leaveId, e.getMessage(), e);
            result.put("success", false);
            result.put("error", e.getMessage());
        }
        return result;
    }

    public Map<String, Object> checkConflicts(String employeeId, LocalDate startDate, LocalDate endDate) {
        Map<String, Object> result = new HashMap<>();
        result.put("success", true);
        result.put("hasConflicts", false);
        List<Map<String, Object>> tasksDue = new ArrayList<>();
        List<Map<String, Object>> teammatesOnLeave = new ArrayList<>();
        List<String> messages = new ArrayList<>();

        if (employeeId == null || employeeId.isBlank() || startDate == null || endDate == null) {
            result.put("tasksDue", tasksDue);
            result.put("teammatesOnLeave", teammatesOnLeave);
            result.put("messages", messages);
            return result;
        }
        if (endDate.isBefore(startDate)) {
            LocalDate tmp = startDate;
            startDate = endDate;
            endDate = tmp;
        }

        Employee employee = employeeRepository.findByEmployeeId(employeeId).orElse(null);
        if (employee == null) {
            result.put("tasksDue", tasksDue);
            result.put("teammatesOnLeave", teammatesOnLeave);
            result.put("messages", messages);
            return result;
        }

        Set<Long> seenTasks = new HashSet<>();
        for (TaskAssignment a : taskAssignmentRepository.findByEmployeeId(employee.getId())) {
            Task t = taskRepository.findById(a.getTaskId()).orElse(null);
            if (t == null || t.getStatus() == TaskStatus.DONE || t.getDueDate() == null) continue;
            if (t.getDueDate().isBefore(startDate) || t.getDueDate().isAfter(endDate)) continue;
            if (!seenTasks.add(t.getId())) continue;
            String projectName = t.getProjectId() != null
                    ? projectRepository.findById(t.getProjectId()).map(Project::getName).orElse("Project #" + t.getProjectId())
                    : "No project";
            Map<String, Object> row = new HashMap<>();
            row.put("taskId", t.getId());
            row.put("title", t.getTitle());
            row.put("dueDate", t.getDueDate().toString());
            row.put("projectName", projectName);
            tasksDue.add(row);
            messages.add("Task \"" + t.getTitle() + "\" is due " + t.getDueDate() + " on " + projectName);
        }

        Set<String> seenTeammates = new HashSet<>();
        for (ProjectMember membership : projectMemberRepository.findByEmployeeId(employee.getId())) {
            Project project = projectRepository.findById(membership.getProjectId()).orElse(null);
            String projectName = project != null ? project.getName() : "Project #" + membership.getProjectId();
            for (ProjectMember other : projectMemberRepository.findByProjectId(membership.getProjectId())) {
                if (other.getEmployeeId().equals(employee.getId())) continue;
                Employee teammate = employeeRepository.findById(other.getEmployeeId()).orElse(null);
                if (teammate == null || teammate.getEmployeeId() == null) continue;
                List<LeaveRequest> leaves = leaveRequestRepository
                        .findApprovedLeavesForEmployeeInRange(teammate.getEmployeeId(), startDate, endDate);
                if (leaves.isEmpty()) continue;
                String key = teammate.getId() + ":" + membership.getProjectId();
                if (!seenTeammates.add(key)) continue;
                String name = ((teammate.getFirstName() == null ? "" : teammate.getFirstName()) + " "
                        + (teammate.getLastName() == null ? "" : teammate.getLastName())).trim();
                if (name.isEmpty()) name = teammate.getEmployeeId();
                LeaveRequest lr = leaves.get(0);
                Map<String, Object> row = new HashMap<>();
                row.put("employeeId", teammate.getEmployeeId());
                row.put("employeeName", name);
                row.put("projectName", projectName);
                row.put("startDate", lr.getStartDate() != null ? lr.getStartDate().toString() : null);
                row.put("endDate", lr.getEndDate() != null ? lr.getEndDate().toString() : null);
                teammatesOnLeave.add(row);
                messages.add(name + " is also on leave during this period (" + projectName + ")");
            }
        }

        boolean has = !tasksDue.isEmpty() || !teammatesOnLeave.isEmpty();
        result.put("hasConflicts", has);
        result.put("tasksDue", tasksDue);
        result.put("teammatesOnLeave", teammatesOnLeave);
        result.put("messages", messages);
        return result;
    }

    private void createAttendanceForApprovedLeave(LeaveRequest leave) {
        try {
            java.time.LocalDate current = leave.getStartDate();
            while (!current.isAfter(leave.getEndDate())) {
                attendanceService.createOrUpdateLeaveAttendance(leave.getEmployeeId(), current, "LEAVE");
                current = current.plusDays(1);
            }
        } catch (Exception e) {
            logger.error("Error creating attendance for approved leave {}: {}", leave.getId(), e.getMessage());
        }
    }

    private void notifyProjectManagersOfLeave(LeaveRequest leave) {
        try {
            if (leave.getStartDate() == null || leave.getEndDate() == null) return;
            Employee emp = employeeRepository.findByEmployeeId(leave.getEmployeeId()).orElse(null);
            if (emp == null) return;

            // Find projects where this person has an open task due inside the leave window
            Set<Long> projectIds = new LinkedHashSet<>();
            for (TaskAssignment a : taskAssignmentRepository.findByEmployeeId(emp.getId())) {
                Task t = taskRepository.findById(a.getTaskId()).orElse(null);
                if (t == null || t.getProjectId() == null || t.getDueDate() == null || t.getStatus() == TaskStatus.DONE) continue;
                if (t.getDueDate().isBefore(leave.getStartDate()) || t.getDueDate().isAfter(leave.getEndDate())) continue;
                projectIds.add(t.getProjectId());
            }
            if (projectIds.isEmpty()) return;

            // Recipients per project: owner + MANAGER/LEAD members + lead of the project's team
            Map<Long, Map<Long, String>> byOwner = new LinkedHashMap<>();
            for (Long pid : projectIds) {
                Project p = projectRepository.findById(pid).orElse(null);
                if (p == null) continue;
                Set<Long> recipients = new LinkedHashSet<>();
                if (p.getOwnerId() != null) recipients.add(p.getOwnerId());
                projectMemberRepository.findByProjectId(pid).stream()
                        .filter(m -> m.getRole() == ProjectRole.MANAGER || m.getRole() == ProjectRole.LEAD)
                        .forEach(m -> recipients.add(m.getEmployeeId()));
                if (p.getTeamId() != null) {
                    teamRepository.findById(p.getTeamId()).map(Team::getLeadEmployeeId)
                            .ifPresent(lead -> { if (lead != null) recipients.add(lead); });
                }
                recipients.remove(emp.getId());
                for (Long r : recipients) {
                    byOwner.computeIfAbsent(r, k -> new LinkedHashMap<>()).put(pid, p.getName());
                }
            }

            String name = ((emp.getFirstName() == null ? "" : emp.getFirstName()) + " "
                    + (emp.getLastName() == null ? "" : emp.getLastName())).trim();
            if (name.isEmpty()) name = emp.getEmployeeId();

            for (Map.Entry<Long, Map<Long, String>> e : byOwner.entrySet()) {
                String ownerCode = employeeRepository.findById(e.getKey())
                        .map(Employee::getEmployeeId).orElse(null);
                if (ownerCode == null) continue;
                Long firstProject = e.getValue().keySet().iterator().next();

                NotificationRequest req = NotificationRequest.builder()
                        .recipientId(ownerCode)
                        .recipientType("EMPLOYEE")
                        .type("LEAVE_IMPACT")
                        .action("LEAVE_IMPACT")
                        .title("Team member on leave: " + name)
                        .message("Employee " + name + " is on approved leave from " + leave.getStartDate()
                                + " to " + leave.getEndDate() + ". Please review their open tasks ("
                                + String.join(", ", e.getValue().values()) + ").")
                        .projectId(firstProject)
                        .priority("MEDIUM")
                        .deepLink("/pm/06-tasks")
                        .build();
                notificationEmitter.emit(req);
            }
        } catch (Exception ex) {
            logger.warn("Leave-impact notification failed for leave {}: {}", leave.getId(), ex.getMessage());
        }
    }

package com.mentis.hrms.service;

import com.mentis.hrms.dto.NotificationRequest;

import org.springframework.messaging.simp.SimpMessagingTemplate;

import java.time.LocalDateTime;

/**
 * Phase 4 â€” unified emitter for HRMS + PM notifications.
 * Persists one row, broadcasts to:
 *   /topic/user/{employeeId}/notifications       (personal bell)
 *   /topic/project/{projectId}/events            (per-project board updates)
 *   /topic/global/events                         (announcements)
 *   /topic/pm/events                             (back-compat for pm-core.js)
 */
@Service
@RequiredArgsConstructor
@Slf4j
public class NotificationEmitter {

    private final NotificationRepository repo;
    private final SimpMessagingTemplate messaging;

    @Transactional
    public Notification emit(NotificationRequest req) {
        Notification n = new Notification();
        n.setRecipientId(req.getRecipientId());
        n.setRecipientType(req.getRecipientType());
        n.setType(req.getType());
        n.setTitle(req.getTitle());
        n.setMessage(req.getMessage());
        n.setSender(req.getActorName());              // existing field
        n.setReferenceId(req.getTaskId() == null ? null : String.valueOf(req.getTaskId()));
        n.setReferenceType(req.getTaskId() == null ? null : "TASK");
        n.setTaskId(req.getTaskId());
        n.setProjectId(req.getProjectId());
        n.setActorId(req.getActorId());
        n.setAction(req.getAction() != null ? req.getAction() : req.getType());
        n.setPriority(req.getPriority());
        n.setDeepLink(req.getDeepLink());
        n.setCreatedAt(LocalDateTime.now());
        n.setReadAt(null);
        n.setPersistent(req.isPersistent());

        Notification saved = repo.save(n);

        broadcast(saved);
        return saved;
    }

    private void broadcast(Notification n) {
        Map<String, Object> payload = new HashMap<>();
        payload.put("id", n.getId());
        payload.put("type", n.getType());
        payload.put("action", n.getAction());
        payload.put("title", n.getTitle());
        payload.put("message", n.getMessage());
        payload.put("taskId", n.getTaskId());
        payload.put("projectId", n.getProjectId());
        payload.put("actorId", n.getActorId());
        payload.put("sender", n.getSender());
        payload.put("priority", n.getPriority());
        payload.put("deepLink", n.getDeepLink());
        payload.put("referenceId", n.getReferenceId());
        payload.put("referenceType", n.getReferenceType());
        payload.put("createdAt", n.getCreatedAt() != null ? n.getCreatedAt().toString() : null);
        payload.put("persistent", n.isPersistent());

        // 1) personal bell â€” keyed by recipientId string (e.g. "SA001")
        try {
            messaging.convertAndSend(
                    "/topic/user/" + n.getRecipientId() + "/notifications", payload);
        } catch (Exception e) {
            log.warn("Per-user broadcast failed for {}: {}", n.getRecipientId(), e.getMessage());
        }

        // 2) per-project board â€” keyed by projectId
        if (n.getProjectId() != null) {
            try {
                messaging.convertAndSend(
                        "/topic/project/" + n.getProjectId() + "/events", payload);
            } catch (Exception e) {
                log.warn("Per-project broadcast failed for {}: {}", n.getProjectId(), e.getMessage());
            }
        }

        // 3) legacy PM topic (keeps pm-core.js working with zero changes)
        try {
            messaging.convertAndSend("/topic/pm/events", payload);
        } catch (Exception e) {
            log.warn("Legacy PM broadcast failed: {}", e.getMessage());
        }
    }
}

    private void sendRealTimeNotification(Notification notification) {
        Map<String, Object> payload = new HashMap<>();
        payload.put("id", notification.getId());
        payload.put("type", notification.getType());
        payload.put("title", notification.getTitle());
        payload.put("message", notification.getMessage());
        payload.put("referenceId", notification.getReferenceId());
        payload.put("sender", notification.getSender());
        payload.put("createdAt", notification.getCreatedAt().toString());
        payload.put("read", notification.getReadAt() != null);
        payload.put("persistent", notification.isPersistent());
        payload.put("color", notification.getNotificationColor());
        payload.put("icon", notification.getNotificationIcon());

        String topic = String.format("/topic/%s/%s/notifications",
                notification.getRecipientType().toLowerCase(),
                notification.getRecipientId());

        messagingTemplate.convertAndSend(topic, payload);
    }
